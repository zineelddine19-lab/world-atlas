        function renderGrid(data) {
            const grid = document.getElementById('countries-grid');
            grid.innerHTML = data.map(item => `
                <div class="card" onclick="showModal('${item.name}')">
                    <div class="flag-box">${item.flag}</div>
                    <div class="card-info">
                        <h3>${item.name}</h3>
                        <p><strong>Capital:</strong> ${item.capital}</p>
                        <p><strong>Region:</strong> ${item.region}</p>
                    </div>
                </div>
            `).join('');
        }

        function filterData() {
            const query = document.getElementById('search').value.toLowerCase();
            const region = document.getElementById('region-filter').value;

            const filtered = countries.filter(c => {
                const matchesSearch = c.name.toLowerCase().includes(query) || c.capital.toLowerCase().includes(query);
                const matchesRegion = region === 'All' || c.region === region;
                return matchesSearch && matchesRegion;
            });

            renderGrid(filtered);
        }

        function showModal(countryName) {
            const c = countries.find(item => item.name === countryName);
            const body = document.getElementById('modal-body');
            body.innerHTML = `
                <span class="close-btn" onclick="closeModal(true)">&times;</span>
                <div class="flag-box" style="height:180px; margin-bottom:15px;">${c.flag}</div>
                <h2>${c.name}</h2>
                <p style="margin-top:10px;"><strong>Capital:</strong> ${c.capital}</p>
                <p><strong>Region:</strong> ${c.region}</p>
                <p><strong>Population:</strong> ${c.pop}</p>
                <p><strong>Currency:</strong> ${c.currency}</p>
            `;
            document.getElementById('modal').classList.add('active');
        }

        function closeModal(e) {
            if (e === true || e.target.id === 'modal') {
                document.getElementById('modal').classList.remove('active');
            }
        }

        function toggleTheme() {
            const current = document.body.getAttribute('data-theme');
            document.body.setAttribute('data-theme', current === 'dark' ? 'light' : 'dark');
        }

        function toggleView(view) {
            document.getElementById('atlas-view').style.display = view === 'atlas' ? 'block' : 'none';
            const quizView = document.getElementById('quiz-view');
            if (view === 'quiz') {
                quizView.classList.add('active');
                nextQuizQuestion();
            } else {
                quizView.classList.remove('active');
            }
        }

        function nextQuizQuestion() {
            const correct = countries[Math.floor(Math.random() * countries.length)];
            document.getElementById('quiz-flag').innerHTML = correct.flag;
            
            let options = [correct.name];
            while(options.length < 4) {
                let randomName = countries[Math.floor(Math.random() * countries.length)].name;
                if(!options.includes(randomName)) options.push(randomName);
            }
            options.sort(() => Math.random() - 0.5);

            const optionsBox = document.getElementById('quiz-options');
            optionsBox.innerHTML = options.map(opt => `
                <button class="option-btn" onclick="checkAnswer('${opt}', '${correct.name}')">${opt}</button>
            `).join('');
        }

        function checkAnswer(selected, correct) {
            if(selected === correct) {
                alert("Correct! 🎉");
                score += 10;
            } else {
                alert(`Wrong! The correct answer was ${correct}.`);
            }
            document.getElementById('quiz-score').innerText = `Score: ${score}`;
            nextQuizQuestion();
        }

        renderGrid(countries);
